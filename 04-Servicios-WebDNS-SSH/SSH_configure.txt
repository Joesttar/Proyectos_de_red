# --- CONFIGURACIÓN PARA RED CONTABILIDAD (192.168.10.0) ---

# 1. Excluir las primeras 10 IPs (para uso estático futuro)
ip dhcp excluded-address 192.168.10.1 192.168.10.10

# 2. Crear el estanque (Pool)
ip dhcp pool POOL-CONTABILIDAD
 network 192.168.10.0 255.255.255.0
 default-router 192.168.10.1
 dns-server 8.8.8.8
 exit
